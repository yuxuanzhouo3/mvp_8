### 第三步：运行修复脚本

如果诊断显示有问题，**复制粘贴以下修复脚本并回车**：

```javascript
(async function() {
  console.log('🔧 开始修复...\n')

  const userInfo = wx.getStorageSync('sitehub_userInfo')
  const db = wx.cloud.database()

  // 生成稳定的userId
  function generateStableUserId(openid) {
    if (!openid) return Math.floor(Math.random() * 900000) + 100000
    let hash = 0
    for (let i = 0; i < openid.length; i++) {
      hash = (hash * 131 + openid.charCodeAt(i)) % 1000000
    }
    return (hash % 900000) + 100000
  }

  const permanentUserId = generateStableUserId(userInfo.openid)
  console.log(`🔑 永久userId: ${permanentUserId}\n`)

  // 1. 查询所有用户记录
  const allUsers = await db.collection('sitehub_users')
    .where({ openid: userInfo.openid })
    .get()

  console.log(`📋 找到 ${allUsers.data.length} 条用户记录`)

  // 2. 选择主记录
  let masterRecord = allUsers.data[0]
  const proRecord = allUsers.data.find(u => u.is_pro === true)
  if (proRecord) {
    masterRecord = proRecord
    console.log('✅ 选择了is_pro=true的记录作为主记录')
  }

  // 3. 查询订阅
  const subscriptions = await db.collection('sitehub_subscriptions')
    .where({ user_openid: userInfo.openid })
    .orderBy('created_at', 'desc')
    .get()

  console.log(`📦 找到 ${subscriptions.data.length} 条订阅记录`)

  // 4. 检查是否有active或pending订阅
  let activeSub = subscriptions.data.find(s => s.status === 'active')
  const pendingSub = subscriptions.data.find(s => s.status === 'pending')

  if (!activeSub && pendingSub) {
    console.log('⚠️ 发现pending订阅，正在激活...')
    await db.collection('sitehub_subscriptions')
      .doc(pendingSub._id)
      .update({
        data: {
          status: 'active',
          updated_at: new Date()
        }
      })
    activeSub = pendingSub
    activeSub.status = 'active'
    console.log('✅ pending订阅已激活')
  }

  // 5. 更新主记录
  const updates = {
    userId: permanentUserId,
    user_id: permanentUserId,
    updated_at: new Date()
  }

  if (activeSub) {
    updates.is_pro = true
    updates.subscription_status = 'active'
    updates.subscription_expires_at = activeSub.current_period_end
    console.log('✅ 设置会员状态: is_pro=true')
  }

  await db.collection('sitehub_users')
    .doc(masterRecord._id)
    .update({ data: updates })

  console.log(`✅ 用户记录已更新\n`)

  // 6. 删除重复记录
  if (allUsers.data.length > 1) {
    console.log('🗑️ 删除重复记录...')
    for (const user of allUsers.data) {
      if (user._id !== masterRecord._id) {
        await db.collection('sitehub_users').doc(user._id).remove()
        console.log(`  ✅ 删除: ${user._id}`)
      }
    }
  }

  // 7. 更新订阅关联
  if (subscriptions.data.length > 0) {
    console.log('🔗 更新订阅关联...')
    for (const sub of subscriptions.data) {
      await db.collection('sitehub_subscriptions')
        .doc(sub._id)
        .update({
          data: {
            user_id: permanentUserId,
            updated_at: new Date()
          }
        })
    }
    console.log(`  ✅ 已更新 ${subscriptions.data.length} 条订阅`)
  }

  // 8. 更新本地缓存
  console.log('💾 更新本地缓存...')
  wx.setStorageSync('sitehub_userInfo', {
    ...userInfo,
    userId: permanentUserId,
    isPro: !!activeSub
  })

  // 9. 验证结果
  console.log('\n✅ 修复完成！')
  console.log(`   永久userId: ${permanentUserId}`)
  console.log(`   会员状态: ${activeSub ? '已激活' : '未激活'}`)
  console.log('\n🔄 请手动刷新页面或重启小程序')
})()
```
