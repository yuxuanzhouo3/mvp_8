# 🔍 诊断和清理支付记录脚本

> 查看你的支付记录，清理测试数据，重新测试支付

## Step 1: 查看当前支付记录和会员状态 (3分钟)

### 在开发者工具控制台执行:

```javascript
(async () => {
  console.log('🔍 开始诊断支付记录和会员状态...\n')

  const userInfo = wx.getStorageSync('sitehub_userInfo')

  if (!userInfo || !userInfo.openid) {
    console.error('❌ 用户未登录')
    return
  }

  console.log('👤 当前用户:', userInfo.nickName)
  console.log('📱 OpenID:', userInfo.openid)
  console.log('━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n')

  // 1. 查看所有订阅记录
  const subsResult = await wx.cloud.database()
    .collection('sitehub_subscriptions')
    .where({ user_openid: userInfo.openid })
    .orderBy('created_at', 'desc')
    .get()

  console.log('📦 订阅记录总数:', subsResult.data.length)

  if (subsResult.data.length > 0) {
    subsResult.data.forEach((sub, index) => {
      console.log(`\n订阅 ${index + 1}:`)
      console.log('  - ID:', sub._id)
      console.log('  - 状态:', sub.status)
      console.log('  - 套餐:', sub.plan_type, '-', sub.billing_cycle)
      console.log('  - 金额:', sub.amount, sub.currency)
      console.log('  - 订单号:', sub.wechat_order_id)
      console.log('  - 交易号:', sub.wechat_transaction_id || '无')
      console.log('  - 创建时间:', new Date(sub.created_at).toLocaleString())
      console.log('  - 到期时间:', sub.current_period_end ? new Date(sub.current_period_end).toLocaleString() : '无')
    })
  } else {
    console.log('⚠️ 没有找到订阅记录')
  }

  console.log('\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n')

  // 2. 查看用户Pro状态
  const userResult = await wx.cloud.database()
    .collection('sitehub_users')
    .where({ openid: userInfo.openid })
    .get()

  if (userResult.data.length > 0) {
    const user = userResult.data[0]
    console.log('👤 用户会员状态:')
    console.log('  - is_pro:', user.is_pro)
    console.log('  - subscription_status:', user.subscription_status || '无')
    console.log('  - subscription_expires_at:', user.subscription_expires_at ? new Date(user.subscription_expires_at).toLocaleString() : '无')
  } else {
    console.log('⚠️ 用户记录未找到')
  }

  console.log('\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n')

  // 3. 查看订单记录
  const ordersResult = await wx.cloud.database()
    .collection('orders')
    .where({ userId: userInfo.openid })
    .orderBy('createdAt', 'desc')
    .get()

  console.log('📝 订单记录总数:', ordersResult.data.length)

  if (ordersResult.data.length > 0) {
    ordersResult.data.forEach((order, index) => {
      console.log(`\n订单 ${index + 1}:`)
      console.log('  - ID:', order._id)
      console.log('  - 订单号:', order.orderNo)
      console.log('  - 状态:', order.status)
      console.log('  - 金额:', order.amount, order.currency)
      console.log('  - 套餐:', order.planType, '-', order.billingCycle)
      console.log('  - 创建时间:', new Date(order.createdAt).toLocaleString())
    })
  } else {
    console.log('⚠️ 没有找到订单记录')
  }

  console.log('\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n')

  // 4. 查看订阅历史
  const historyResult = await wx.cloud.database()
    .collection('sitehub_subscription_history')
    .where({ user_openid: userInfo.openid })
    .orderBy('created_at', 'desc')
    .limit(10)
    .get()

  console.log('📊 订阅历史记录:', historyResult.data.length)

  if (historyResult.data.length > 0) {
    historyResult.data.forEach((history, index) => {
      console.log(`\n历史 ${index + 1}:`)
      console.log('  - 操作:', history.action)
      console.log('  - 金额:', history.amount)
      console.log('  - 订阅ID:', history.subscription_id)
      console.log('  - 交易号:', history.transaction_id)
      console.log('  - 时间:', new Date(history.created_at).toLocaleString())
      console.log('  - 备注:', history.notes)
    })
  } else {
    console.log('⚠️ 没有找到订阅历史')
  }

  console.log('\n━━━━━━━━━━━━━━━━━━━━━━━━━━━━')
  console.log('🎉 诊断完成！')
  console.log('\n如需清理测试数据，请执行 Step 2 的脚本')
})()
```

## Step 2: 清除所有测试数据 (1分钟)

### ⚠️ 警告: 这将删除你的所有支付测试数据！

```javascript
(async () => {
  console.log('🔧 开始清除测试数据...\n')

  const userInfo = wx.getStorageSync('sitehub_userInfo')

  if (!userInfo || !userInfo.openid) {
    console.error('❌ 用户未登录')
    return
  }

  // 确认操作
  const confirmResult = await new Promise((resolve) => {
    wx.showModal({
      title: '确认清除',
      content: `即将清除用户 ${userInfo.nickName} 的所有测试数据：\n\n• 所有订阅记录\n• 订阅历史\n• 订单记录\n• 会员状态\n\n确定要继续吗？`,
      confirmText: '确定清除',
      cancelText: '取消',
      success: (res) => resolve(res.confirm)
    })
  })

  if (!confirmResult) {
    console.log('❌ 用户取消清除操作')
    return
  }

  console.log('🗑️ 调用云函数清除数据...')

  try {
    const result = await wx.cloud.callFunction({
      name: 'wechatPaySubscription',
      data: {
        action: 'clearTestData',
        userInfo: userInfo
      }
    })

    console.log('✅ 清除完成:', result)
    console.log('\n📊 清除结果:')
    console.log('  - 订阅记录删除:', result.result.data.subscriptionsRemoved, '条')
    console.log('  - 订阅历史删除:', result.result.data.historyRemoved, '条')
    console.log('  - 用户状态更新:', result.result.data.usersUpdated, '个')
    console.log('  - 订单状态更新:', result.result.data.ordersUpdated, '个')
    console.log('\n✅', result.result.message)

    // 清除本地缓存的会员状态
    userInfo.isPro = false
    userInfo.pro = false
    wx.setStorageSync('sitehub_userInfo', userInfo)
    console.log('📱 本地缓存已更新')

    // 提示用户
    wx.showModal({
      title: '清除成功',
      content: '测试数据已清除完毕！\n\n现在可以重新测试支付流程了。',
      showCancel: false,
      confirmText: '知道了'
    })

    console.log('\n🎉 现在可以重新测试支付了！')

  } catch (error) {
    console.error('❌ 清除失败:', error)
    wx.showToast({
      title: '清除失败',
      icon: 'none'
    })
  }
})()
```

## Step 3: 验证清除结果 (1分钟)

```javascript
(async () => {
  console.log('🔍 验证清除结果...\n')

  const userInfo = wx.getStorageSync('sitehub_userInfo')

  // 1. 检查订阅记录
  const subsResult = await wx.cloud.database()
    .collection('sitehub_subscriptions')
    .where({ user_openid: userInfo.openid })
    .get()

  console.log('📦 订阅记录数:', subsResult.data.length, subsResult.data.length === 0 ? '✅' : '❌')

  // 2. 检查用户状态
  const userResult = await wx.cloud.database()
    .collection('sitehub_users')
    .where({ openid: userInfo.openid })
    .get()

  if (userResult.data.length > 0) {
    const user = userResult.data[0]
    console.log('👤 is_pro:', user.is_pro, user.is_pro === false ? '✅' : '❌')
    console.log('👤 subscription_status:', user.subscription_status, user.subscription_status === 'inactive' ? '✅' : '❌')
  }

  // 3. 检查本地缓存
  console.log('📱 本地 isPro:', userInfo.isPro, userInfo.isPro === false ? '✅' : '❌')

  console.log('\n✅ 验证完成！现在是全新状态，可以重新测试支付了。')
})()
```

## 🔧 关于两个云函数

### wechatPay vs wechatPaySubscription

**wechatPay** (旧版):
- ❌ 功能简单，只创建订单
- ❌ 没有订阅管理
- ❌ 没有状态更新
- ❌ 没有清理功能
- **建议**: 可以删除

**wechatPaySubscription** (新版):
- ✅ 完整的订阅管理系统
- ✅ 订阅创建、查询、取消
- ✅ 支付回调处理
- ✅ 订阅历史记录
- ✅ 清除测试数据功能 ⭐
- ✅ 完善的错误处理
- **建议**: 使用这个

### 删除 wechatPay 云函数

如果确认不需要，可以删除:
```bash
1. 在微信开发者工具中
2. 右键 cloudfunctions/wechatPay
3. 选择"删除"
4. 确认删除
```

## 📝 常见问题

### Q1: 为什么会有两次会员记录？
可能是测试时多次支付导致的，使用 Step 2 清除所有测试数据即可。

### Q2: 清除数据会影响什么？
只会清除测试环境的数据:
- ✅ 删除所有订阅记录
- ✅ 删除订阅历史
- ✅ 重置会员状态为 false
- ✅ 订单状态改为 cancelled
- ⚠️ 不会删除用户账号

### Q3: 支付成功但没有账单显示？
需要在设置页面添加账单列表UI，我会在下一步创建UI设计方案。

### Q4: 清除后能恢复吗？
❌ 不能，测试数据会永久删除。如果需要保留，请先导出数据。
