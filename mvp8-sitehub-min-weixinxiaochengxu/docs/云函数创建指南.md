### 3.3 创建云函数代码

将以下代码复制到 `cloudfunctions/wechatPay/index.js`：

\`\`\`javascript
// 云函数入口文件
const cloud = require('wx-server-sdk')
const axios = require('axios')
const crypto = require('crypto')

cloud.init({
  env: cloud.DYNAMIC_CURRENT_ENV
})

const db = cloud.database()

// 微信支付商户配置
const WECHAT_PAY_CONFIG = {
  appid: 'wxf1aca21b5b79581d',
  mchid: '1694786758',
  apiKey: 'mornsciencemornscience2025100705',
  notifyUrl: 'https://your-domain.com/api/wechat/pay/notify' // 需要替换成实际回调地址
}

// 生成随机字符串
function generateNonceStr() {
  return Math.random().toString(36).substr(2, 15)
}

// 生成签名
function generateSign(params, apiKey) {
  // 按字典序排序
  const sortedKeys = Object.keys(params).sort()
  const stringA = sortedKeys
    .map(key => \`\${key}=\${params[key]}\`)
    .join('&')
  const stringSignTemp = \`\${stringA}&key=\${apiKey}\`

  // MD5 签名
  return crypto
    .createHash('md5')
    .update(stringSignTemp)
    .digest('hex')
    .toUpperCase()
}

// 云函数入口函数
exports.main = async (event, context) => {
  const wxContext = cloud.getWXContext()
  const { planType, billingCycle, amount } = event

  try {
    // 生成订单号
    const outTradeNo = \`SITEHUB_\${Date.now()}_\${Math.random().toString(36).substr(2, 9)}\`

    // 计算金额（单位：分）
    const totalFee = amount * 100

    // 构建统一下单参数
    const unifiedOrderParams = {
      appid: WECHAT_PAY_CONFIG.appid,
      mch_id: WECHAT_PAY_CONFIG.mchid,
      nonce_str: generateNonceStr(),
      body: \`SiteHub Pro会员-\${billingCycle === 'monthly' ? '月付' : '年付'}\`,
      out_trade_no: outTradeNo,
      total_fee: totalFee,
      spbill_create_ip: '127.0.0.1',
      notify_url: WECHAT_PAY_CONFIG.notifyUrl,
      trade_type: 'JSAPI',
      openid: wxContext.OPENID
    }

    // 生成签名
    unifiedOrderParams.sign = generateSign(unifiedOrderParams, WECHAT_PAY_CONFIG.apiKey)

    // 转换为 XML
    const xmlParams = Object.keys(unifiedOrderParams)
      .map(key => \`<\${key}>\${unifiedOrderParams[key]}</\${key}>\`)
      .join('')
    const xmlData = \`<xml>\${xmlParams}</xml>\`

    // 调用微信统一下单接口
    const unifiedOrderRes = await axios.post(
      'https://api.mch.weixin.qq.com/pay/unifiedorder',
      xmlData,
      {
        headers: {
          'Content-Type': 'text/xml'
        }
      }
    )

    // 解析返回的 XML（简化处理，实际应使用 XML 解析库）
    const prepayId = unifiedOrderRes.data.match(/<prepay_id><!\[CDATA\[(.*?)\]\]><\/prepay_id>/)?.[1]

    if (!prepayId) {
      throw new Error('获取 prepay_id 失败')
    }

    // 构建小程序支付参数
    const paymentParams = {
      appId: WECHAT_PAY_CONFIG.appid,
      timeStamp: Math.floor(Date.now() / 1000).toString(),
      nonceStr: generateNonceStr(),
      package: \`prepay_id=\${prepayId}\`,
      signType: 'MD5'
    }

    // 生成支付签名
    paymentParams.paySign = generateSign(paymentParams, WECHAT_PAY_CONFIG.apiKey)

    // 保存订单到数据库
    await db.collection('orders').add({
      data: {
        orderNo: outTradeNo,
        userId: wxContext.OPENID,
        planType,
        billingCycle,
        amount,
        status: 'pending',
        prepayId: prepayId,
        createdAt: db.serverDate(),
        expiresAt: new Date(Date.now() + 2 * 60 * 60 * 1000) // 2小时过期
      }
    })

    return {
      success: true,
      payment: paymentParams
    }
  } catch (error) {
    console.error('微信支付错误:', error)
    return {
      success: false,
      message: error.message || '支付创建失败'
    }
  }
}

## ⚠️ 重要提示

### 1. 回调地址配置

`notifyUrl` 需要是一个**公网可访问的 HTTPS 地址**。您有两个选择：

**选项 A**: 使用云函数 HTTP 触发器
- 在云开发控制台中，为云函数开启 HTTP 触发
- 获取云函数 HTTP 访问路径
- 将该路径设置为 `notifyUrl`

**选项 B**: 使用您的网站后端
- 使用 `https://www.mornhub.help/api/wechat/pay/notify`
- 在网站后端创建对应的回调接口

### 2. 微信支付 APIv3 升级建议

当前代码使用的是 APIv2（MD5 签名）。微信官方推荐升级到 APIv3（证书签名）：

- 更安全
- 更完善的功能
- 官方长期支持

如需升级，需要在微信商户平台申请 APIv3 密钥和证书。

### 3. 测试环境配置

1. 设置支付授权目录
2. 添加测试白名单（您的微信号）
