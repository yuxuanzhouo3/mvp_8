(async()=>{try{console.log('🔧 清除...');const u=wx.getStorageSync('sitehub_userInfo');if(!u){console.error('❌ 未登录');return}const r=await wx.cloud.callFunction({name:'wechatPaySubscription',data:{action:'clearTestData',userInfo:u}});console.log('📊',r.result);if(r.result.success){console.log('✅ 已清除:',r.result.data);u.pro=false;u.isPro=false;wx.setStorageSync('sitehub_userInfo',u);const a=getApp();if(a.globalData.userInfo){a.globalData.userInfo.pro=false;a.globalData.userInfo.isPro=false}console.log('💡 可以测试了');setTimeout(()=>wx.reLaunch({url:'/pages/settings/settings'}),2000)}else{console.error('❌',r.result.error)}}catch(e){console.error('❌',e)}})()

### 3️⃣ 测试支付（验证）

**复制到真机控制台运行：**

```javascript
(async()=>{try{console.log('🧪 测试...');const u=wx.getStorageSync('sitehub_userInfo');if(!u){console.error('❌ 未登录');return}const r=await wx.cloud.callFunction({name:'wechatPaySubscription',data:{action:'createSubscription',planType:'personal',billingCycle:'yearly',amount:168,userInfo:u}});console.log('📊',r.result);if(r.result.success&&r.result.data){const{orderId,orderNo,paymentParams}=r.result.data;console.log('✅ 字段:',{orderId:!!orderId,orderNo:!!orderNo,paymentParams:!!paymentParams});if(paymentParams){console.log('✅ 参数:',{timeStamp:!!paymentParams.timeStamp,nonceStr:!!paymentParams.nonceStr,package:!!paymentParams.package,signType:!!paymentParams.signType,paySign:!!paymentParams.paySign});if(paymentParams.timeStamp&&paymentParams.paySign){console.log('🎉 完整！')}}}else{console.error('❌',r.result.error)}}catch(e){console.error('❌',e)}})()
```
