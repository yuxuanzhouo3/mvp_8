### 支付成功回调链路 (pages/payment/payment.js: Line 303-385)

```javascript
async handlePaymentSuccess(orderId, userInfo, payRes) {
  // 1️⃣ 更新订单状态为已支付
  await wx.cloud.database()
    .collection('orders')
    .where({ orderNo: orderId })
    .update({
      data: {
        status: 'paid',
        transaction_id: transactionId,
        paidAt: new Date(),
        updatedAt: new Date()
      }
    })

  // 2️⃣ 创建订阅记录（状态为 active）
  const subscriptionData = {
    user_openid: userInfo.openid,
    plan_type: this.data.planType,
    billing_cycle: this.data.billingCycle,
    status: 'active',  // ✅ 直接设置为 active
    auto_renew: true,
    wechat_order_id: orderId,
    wechat_transaction_id: transactionId,
    amount: this.data.amount,
    currency: 'CNY',
    start_date: new Date(),
    current_period_end: this.calculatePeriodEnd(new Date(), this.data.billingCycle),
    cancel_at_period_end: false,
    created_at: new Date(),
    updated_at: new Date(),
    version: Date.now()
  }

  await wx.cloud.database()
    .collection('sitehub_subscriptions')
    .add({ data: subscriptionData })

  // 3️⃣ 更新用户会员状态
  await wx.cloud.database()
    .collection('sitehub_users')
    .where({ openid: userInfo.openid })
    .update({
      data: {
        is_pro: true,  // ✅ 设置为 Pro
        subscription_status: 'active',
        subscription_expires_at: subscriptionData.current_period_end,
        version: Date.now()
      }
    })

  // 4️⃣ 更新本地缓存
  this.updateLocalUserStatus(true)

  // 5️⃣ 更新全局状态
  const app = getApp()
  if (app.globalData.userInfo) {
    app.globalData.userInfo.pro = true
    app.globalData.userInfo.isPro = true
  }

  // 6️⃣ 强制刷新用户画像
  const { getUserSession } = require('../../utils/userSession')
  const userSession = getUserSession()
  await userSession.fetchMe(true)  // ✅ 强制刷新，不使用缓存
}
```

### 问题根因的本质

不是代码bug，而是**类型转换的隐式假设**:
- 前端假设字符串会自动转换
- 云函数假设 `Number()` 总是成功
- 微信API不接受类型不匹配的参数

**解决方案**: 显式转换 + 严格验证 + 详细日志
