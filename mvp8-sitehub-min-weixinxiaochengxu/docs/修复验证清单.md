## 📞 快速诊断命令

### 一键检查
```javascript
// 复制粘贴到微信开发者工具控制台
(async () => {
  console.log('🔍 开始系统检查...\n')

  // 1. 检查云函数
  try {
    const config = await wx.cloud.callFunction({
      name: 'wechatPaySubscription',
      data: { action: 'checkConfig' }
    })
    console.log('✅ 云函数连通:', config.result.data.isConfigured ? '正常' : '需配置')
    console.log('   - AppID:', config.result.data.appid)
    console.log('   - 商户号:', config.result.data.mchId)
    console.log('   - API密钥长度:', config.result.data.apiKeyLength)
  } catch (e) {
    console.error('❌ 云函数异常:', e.message)
  }

  // 2. 检查用户登录
  const userInfo = wx.getStorageSync('sitehub_userInfo')
  if (userInfo && userInfo.openid) {
    console.log('\n✅ 用户已登录:', userInfo.nickName)
    console.log('   - openid:', userInfo.openid)
  } else {
    console.error('\n❌ 用户未登录')
  }

  // 3. 检查数据库
  try {
    const subs = await wx.cloud.database().collection('sitehub_subscriptions').count()
    console.log('\n✅ 订阅记录数:', subs.total)

    const users = await wx.cloud.database().collection('sitehub_users').count()
    console.log('✅ 用户记录数:', users.total)

    const pricing = await wx.cloud.database().collection('sitehub_pricing').count()
    console.log('✅ 价格配置数:', pricing.total)
  } catch (e) {
    console.error('\n❌ 数据库异常:', e.message)
  }

  console.log('\n🎉 系统检查完成！')
})()
```
